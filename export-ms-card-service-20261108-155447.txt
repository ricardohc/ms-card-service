package com.banking.card.client;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.core.publisher.Mono;

import java.util.UUID;

/** Validates linked account existence and ACTIVE status against ms-account-service. */
@Slf4j
@Component
@RequiredArgsConstructor
public class AccountValidationClient {

    private final WebClient.Builder webClientBuilder;

    @Value("${external.services.account.url:http://ms-account-service}")
    private String accountServiceUrl;

    /** Returns true when the account exists and is ACTIVE. */
    public Mono<Boolean> accountActive(UUID accountId) {
        return webClientBuilder.build()
                .get()
                .uri(accountServiceUrl + "/api/v1/accounts/{accountId}", accountId)
                .header("X-Correlation-ID", UUID.randomUUID().toString())
                .header("X-Channel-ID", "MS-CARD-SERVICE")
                .header("Accept", "application/json")
                .retrieve()
                .bodyToMono(String.class)
                .map(body -> body != null && body.contains("ACTIVE"))
                .onErrorResume(ex -> {
                    log.warn("Account validation failed for {}: {}", accountId, ex.getMessage());
                    return Mono.just(false);
                });
    }
}

package com.banking.card.client;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.core.publisher.Mono;

import java.util.UUID;

/** Validates customer existence against ms-customer-service via its HTTP API. */
@Slf4j
@Component
@RequiredArgsConstructor
public class CustomerValidationClient {

    private final WebClient.Builder webClientBuilder;

    @Value("${external.services.customer.url:http://ms-customer-service}")
    private String customerServiceUrl;

    /** Returns true when the customer exists in ms-customer-service. */
    public Mono<Boolean> customerExists(UUID customerId) {
        return webClientBuilder.build()
                .get()
                .uri(customerServiceUrl + "/api/v1/customers/{customerId}", customerId)
                .header("X-Correlation-ID", UUID.randomUUID().toString())
                .header("X-Channel-ID", "MS-CARD-SERVICE")
                .header("Accept", "application/json")
                .retrieve()
                .toBodilessEntity()
                .map(response -> true)
                .onErrorResume(ex -> {
                    log.warn("Customer validation failed for {}: {}", customerId, ex.getMessage());
                    return Mono.just(false);
                });
    }
}

package com.banking.card.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.annotation.Profile;
import org.springframework.security.config.annotation.web.reactive.EnableWebFluxSecurity;
import org.springframework.security.config.web.server.ServerHttpSecurity;
import org.springframework.security.web.server.SecurityWebFilterChain;

/**
 * OAuth2 Resource Server with JWT (Parte III contract).
 * A permitAll chain exists ONLY under the dev profile while no identity provider
 * is available; production profiles require valid JWTs.
 */
@Configuration
@EnableWebFluxSecurity
public class SecurityConfig {

    @Bean
    @Profile("dev")
    public SecurityWebFilterChain devSecurityWebFilterChain(ServerHttpSecurity http) {
        return http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .authorizeExchange(exchanges -> exchanges.anyExchange().permitAll())
                .build();
    }

    @Bean
    @Profile("!dev")
    public SecurityWebFilterChain jwtSecurityWebFilterChain(ServerHttpSecurity http) {
        return http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .authorizeExchange(exchanges -> exchanges.anyExchange().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2.jwt(jwt -> { }))
                .build();
    }
}

package com.banking.card.controller;

import com.banking.card.api.CardApi;
import com.banking.card.dto.BalanceResponse;
import com.banking.card.dto.CardCreateRequest;
import com.banking.card.dto.CardPage;
import com.banking.card.dto.CardResponse;
import com.banking.card.dto.CardUpdateRequest;
import com.banking.card.dto.ConsumptionRequest;
import com.banking.card.dto.EligibilityResponse;
import com.banking.card.dto.MovementListResponse;
import com.banking.card.dto.OperationAcceptedResponse;
import com.banking.card.service.CardService;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

import java.util.UUID;

/** Thin adapter: generated CardApi -> CardService. No business logic here. */
@RestController
@RequiredArgsConstructor
public class CardApiImpl implements CardApi {

    private final CardService cardService;

    @Override
    public Mono<ResponseEntity<CardPage>> findAllCards(
            String xCorrelationID, String xChannelID, Integer page, Integer size,
            ServerWebExchange exchange) {
        return cardService.findAllCards(page, size)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<CardResponse>> createCard(
            String xCorrelationID, String xChannelID, UUID xTransactionID,
            Mono<CardCreateRequest> cardCreateRequest, ServerWebExchange exchange) {
        return cardCreateRequest
                .flatMap(request -> cardService.createCard(request, xTransactionID))
                .map(response -> ResponseEntity.status(HttpStatus.CREATED).body(response));
    }

    @Override
    public Mono<ResponseEntity<CardResponse>> findCardById(
            UUID cardId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return cardService.findCardById(cardId)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<CardResponse>> updateCard(
            UUID cardId, String xCorrelationID, String xChannelID,
            UUID xTransactionID, Mono<CardUpdateRequest> cardUpdateRequest,
            ServerWebExchange exchange) {
        return cardUpdateRequest
                .flatMap(request -> cardService.updateCard(cardId, request, xTransactionID))
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<Void>> deactivateCard(
            UUID cardId, String xCorrelationID, String xChannelID,
            UUID xTransactionID, ServerWebExchange exchange) {
        return cardService.deactivateCard(cardId, xTransactionID)
                .thenReturn(ResponseEntity.<Void>noContent().build());
    }

    @Override
    public Mono<ResponseEntity<OperationAcceptedResponse>> createConsumption(
            UUID cardId, String xCorrelationID, String xChannelID,
            UUID xTransactionID, Mono<ConsumptionRequest> consumptionRequest,
            ServerWebExchange exchange) {
        return consumptionRequest
                .flatMap(request -> cardService.createConsumption(cardId, request, xTransactionID))
                .map(response -> ResponseEntity.status(HttpStatus.ACCEPTED).body(response));
    }

    @Override
    public Mono<ResponseEntity<BalanceResponse>> findCardBalance(
            UUID cardId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return cardService.findCardBalance(cardId)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<MovementListResponse>> findLastMovements(
            UUID cardId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return cardService.findLastMovements(cardId)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<EligibilityResponse>> checkCardEligibility(
            UUID customerId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return cardService.checkCardEligibility(customerId)
                .map(ResponseEntity::ok);
    }
}

=== src/main/java/com/banking/card/event/CardEventProducer.java ===
package com.banking.card.event;

import com.banking.card.document.CardDocument;
import com.banking.card.document.MovementDocument;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.kafka.core.reactive.ReactiveKafkaProducerTemplate;
import org.springframework.stereotype.Component;
import reactor.core.publisher.Mono;

import java.util.Map;
import java.util.UUID;

/**
 * Produces domain events on topic banking.card.events.v1.
 * Events: CardIssued, CardUpdated, CardCancelled, CardMovementRegistered.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class CardEventProducer {

    public static final String TOPIC = "banking.card.events.v1";

    private final ReactiveKafkaProducerTemplate<String, Object> kafkaTemplate;

    public Mono<Void> cardIssued(CardDocument card) {
        return send("CardIssued", card.getId(), baseEvent(card, "CardIssued"));
    }

    public Mono<Void> cardUpdated(CardDocument card) {
        return send("CardUpdated", card.getId(), baseEvent(card, "CardUpdated"));
    }

    public Mono<Void> cardCancelled(CardDocument card) {
        return send("CardCancelled", card.getId(), baseEvent(card, "CardCancelled"));
    }

    public Mono<Void> movementRegistered(MovementDocument movement) {
        Map<String, Object> event = Map.of(
                "eventType", "CardMovementRegistered",
                "cardId", String.valueOf(movement.getCardId()),
                "movementId", String.valueOf(movement.getId()),
                "type", movement.getType(),
                "amount", movement.getAmount().toPlainString(),
                "timestamp", movement.getTimestamp().toString());
        return send("CardMovementRegistered", movement.getCardId(), event);
    }

    private Map<String, Object> baseEvent(CardDocument card, String eventType) {
        return Map.of(
                "eventType", eventType,
                "cardId", String.valueOf(card.getId()),
                "customerId", String.valueOf(card.getCustomerId()),
                "type", card.getType(),
                "status", card.getStatus());
    }

    private Mono<Void> send(String eventType, UUID key, Object payload) {
        return kafkaTemplate.send(TOPIC, String.valueOf(key), payload)
                .doOnSuccess(result -> log.info("Event {} sent for key {}", eventType, key))
                .onErrorResume(ex -> {
                    // Event publishing must not break the main flow; log and continue.
                    log.error("Failed to send event {} for key {}: {}", eventType, key, ex.getMessage());
                    return Mono.empty();
                })
                .then();
    }
}

package com.banking.card.exception;

/** Base exception carrying the HTTP status and error code of the contract. */
public class DomainException extends RuntimeException {

    private final int status;
    private final String code;

    public DomainException(int status, String code, String message) {
        super(message);
        this.status = status;
        this.code = code;
    }

    public DomainException(int status, String code, String message, Throwable cause) {
        super(message, cause);
        this.status = status;
        this.code = code;
    }

    public int getStatus() {
        return status;
    }

    public String getCode() {
        return code;
    }
}

/** Maps domain exceptions to the ErrorResponse contract of the OpenAPI spec. */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(DomainException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleDomain(DomainException ex) {
        log.warn("Domain error code={} message={}", ex.getCode(), ex.getMessage());
        return Mono.just(build(HttpStatus.valueOf(ex.getStatus()), ex.getCode(), ex.getMessage(), List.of(ex.getMessage())));
    }

    @ExceptionHandler(ServerWebInputException.class)
    public Mono<ResponseEntity<ErrorResponse>> handleInput(ServerWebInputException ex) {
        return Mono.just(build(HttpStatus.BAD_REQUEST, "INVALID_REQUEST", ex.getReason(), List.of(ex.getReason())));
    }

    @ExceptionHandler(Exception.class)
    public Mono<ResponseEntity<ErrorResponse>> handleGeneric(Exception ex) {
        log.error("Unexpected error", ex);
        return Mono.just(build(HttpStatus.INTERNAL_SERVER_ERROR, "INTERNAL_ERROR",
                "Unexpected internal error", List.of(ex.getClass().getSimpleName())));
    }

    private ResponseEntity<ErrorResponse> build(HttpStatus status, String code, String message, List<String> details) {
        ErrorResponse error = new ErrorResponse();
        error.setCode(code);
        error.setMessage(message);
        error.setTimestamp(OffsetDateTime.now());
        error.setCorrelationId(null);
        error.setDetails(details);
        return ResponseEntity.status(status).body(error);
    }
}

package com.banking.card.repository;

import com.banking.card.document.CardDocument;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.bson.types.Decimal128;
import org.springframework.data.mongodb.core.ReactiveMongoTemplate;
import org.springframework.data.mongodb.core.query.Criteria;
import org.springframework.data.mongodb.core.query.Query;
import org.springframework.data.mongodb.core.query.Update;
import org.springframework.stereotype.Repository;
import reactor.core.publisher.Mono;

import java.math.BigDecimal;
import java.util.UUID;

/** Mongo-specific atomic operations that cannot be expressed as derived methods. */
@Slf4j
@Repository
@RequiredArgsConstructor
public class CardAtomicOperations {

    private final ReactiveMongoTemplate mongoTemplate;

    /** Atomically decrements availableBalance only when the current balance covers the amount.
     *  Returns true when the decrement happened, false otherwise. */
    public Mono<Boolean> decrementAvailableBalance(UUID cardId, BigDecimal amount) {
        Decimal128 delta = new Decimal128(amount.negate());
        Decimal128 min = new Decimal128(amount);

        Query query = new Query(new Criteria().andOperator(
                Criteria.where("_id").is(cardId),
                Criteria.where("availableBalance").gte(min)));

        Update update = new Update().inc("availableBalance", delta);

        return mongoTemplate.findAndModify(query, update, CardDocument.class)
                .map(updated -> {
                    log.debug("availableBalance decremented by {} for card {}", amount, cardId);
                    return true;
                })
                .defaultIfEmpty(false)
                .doOnError(ex -> log.error(
                        "Atomic decrement failed for card {} amount {}: {}",
                        cardId, amount, ex.getMessage()));
    }
}

import com.banking.card.client.AccountValidationClient;
import com.banking.card.client.CustomerValidationClient;

/**
 * Reactor-based implementation of card business rules.
 * All cross-service validations go through typed clients (HTTP APIs only);
 * this service never reads another service's database.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class CardServiceImpl implements CardService {

    private static final String STATUS_ACTIVE = CardStatus.ACTIVE.getValue();
    private static final String STATUS_CANCELLED = CardStatus.CANCELLED.getValue();
    private static final String TYPE_CREDIT = CardType.CREDIT.getValue();
    private static final String TYPE_DEBIT = CardType.DEBIT.getValue();
    private static final String DEFAULT_CURRENCY = "PEN";

    private final CardRepository cardRepository;
    private final MovementRepository movementRepository;
    private final CardAtomicOperations atomic;
    private final CardMapper mapper;
    private final CustomerValidationClient customerClient;
    private final AccountValidationClient accountClient;
    private final CardEventProducer eventProducer;

    @Override
    public Mono<CardPage> findAllCards(int page, int size) {
        Pageable pageable = PageRequest.of(page, size, Sort.by(Sort.Direction.DESC, "createdAt"));
        Mono<List<CardResponse>> content = cardRepository.findAllBy(pageable)
                .map(mapper::toResponse)
                .collectList();
        Mono<Long> total = cardRepository.countByStatusNot(STATUS_CANCELLED);
        return Mono.zip(content, total)
                .map(tuple -> {
                    CardPage result = new CardPage();
                    result.setContent(tuple.getT1());
                    result.setPage(page);
                    result.setSize(size);
                    result.setTotalElements(tuple.getT2().intValue());
                    return result;
                });
    }

    @Override
    public Mono<CardResponse> createCard(CardCreateRequest request, UUID transactionId) {
        return Mono.defer(() -> {
            if (request instanceof CreditCardCreateRequest credit) {
                return createCreditCard(credit);
            } else if (request instanceof DebitCardCreateRequest debit) {
                return createDebitCard(debit);
            }
            return Mono.error(new InvalidCardTypeException("Card payload does not match oneOf CREDIT/DEBIT"));
        });
    }

    private Mono<CardResponse> createCreditCard(CreditCardCreateRequest credit) {
        BigDecimal limit = new BigDecimal(credit.getCreditLimit());
        if (limit.signum() <= 0) {
            return Mono.error(new InvalidCardTypeException(
                    "creditLimit must be greater than 0 for CREDIT cards"));
        }
        return customerClient.customerExists(credit.getCustomerId())
                .filter(Boolean::booleanValue)
                .switchIfEmpty(Mono.error(new AccountValidationException("Customer does not exist")))
                .then(duplicateCardCheck(credit.getCustomerId(), TYPE_CREDIT))
                .then(Mono.defer(() -> {
                    CardDocument card = CardDocument.builder()
                            .id(UUID.randomUUID())
                            .customerId(credit.getCustomerId())
                            .type(TYPE_CREDIT)
                            .status(STATUS_ACTIVE)
                            .maskedPan(generateMaskedPan())
                            .creditLimit(limit)
                            .availableBalance(limit)
                            .currency(DEFAULT_CURRENCY)
                            .expirationDate(LocalDate.now(java.time.ZoneOffset.UTC).plusYears(4))
                            .build();
                    return cardRepository.save(card)
                            .flatMap(saved -> eventProducer.cardIssued(saved).thenReturn(saved))
                            .map(mapper::toResponse);
                }));
    }

    private Mono<CardResponse> createDebitCard(DebitCardCreateRequest debit) {
        return customerClient.customerExists(debit.getCustomerId())
                .filter(Boolean::booleanValue)
                .switchIfEmpty(Mono.error(new AccountValidationException("Customer does not exist")))
                .then(accountClient.accountActive(debit.getLinkedAccountId()))
                .filter(Boolean::booleanValue)
                .switchIfEmpty(Mono.error(new AccountValidationException(
                        "linkedAccountId does not exist or is not ACTIVE")))
                .then(Mono.defer(() -> {
                    CardDocument card = CardDocument.builder()
                            .id(UUID.randomUUID())
                            .customerId(debit.getCustomerId())
                            .type(TYPE_DEBIT)
                            .status(STATUS_ACTIVE)
                            .maskedPan(generateMaskedPan())
                            .linkedAccountId(debit.getLinkedAccountId())
                            .currency(DEFAULT_CURRENCY)
                            .expirationDate(LocalDate.now(java.time.ZoneOffset.UTC).plusYears(4))
                            .build();
                    return cardRepository.save(card)
                            .flatMap(saved -> eventProducer.cardIssued(saved).thenReturn(saved))
                            .map(mapper::toResponse);
                }));
    }

    private Mono<Void> duplicateCardCheck(UUID customerId, String type) {
        return cardRepository.existsByCustomerIdAndTypeAndStatus(customerId, type, STATUS_ACTIVE)
                .flatMap(exists -> exists
                        ? Mono.error(new DuplicateCardException("Customer already has an ACTIVE credit card"))
                        : Mono.<Void>empty());
    }

    private String generateMaskedPan() {
        String last4 = String.format("%04d", (int) (Math.random() * 10000));
        return "**** **** **** " + last4;
    }

    @Override
    public Mono<CardResponse> findCardById(UUID cardId) {
        return getActiveCard(cardId).map(mapper::toResponse);
    }

    @Override
    public Mono<CardResponse> updateCard(UUID cardId, CardUpdateRequest request, UUID transactionId) {
        return getActiveCard(cardId)
                .flatMap(card -> {
                    if (request.getStatus() != null) {
                        String target = request.getStatus().getValue();
                        if (STATUS_CANCELLED.equals(target)) {
                            return Mono.error(new InvalidCardStateException(
                                    "Use DELETE for definitive cancellation"));
                        }
                        card.setStatus(target);
                    }
                    return cardRepository.save(card)
                            .flatMap(saved -> eventProducer.cardUpdated(saved).thenReturn(saved))
                            .map(mapper::toResponse);
                });
    }

    @Override
    public Mono<Void> deactivateCard(UUID cardId, UUID transactionId) {
        return getActiveCard(cardId)
                .flatMap(card -> {
                    card.setStatus(STATUS_CANCELLED);
                    return cardRepository.save(card)
                            .flatMap(saved -> eventProducer.cardCancelled(saved).then());
                });
    }

    @Override
    public Mono<OperationAcceptedResponse> createConsumption(UUID cardId, ConsumptionRequest request, UUID transactionId) {
        return Mono.defer(() -> {
            BigDecimal amount = new BigDecimal(request.getAmount());

            // 1. VALIDACIÓN TEMPRANA: El monto debe ser mayor a cero (Evita que llegue a Mongo o Kafka)
            if (amount.compareTo(BigDecimal.ZERO) <= 0) {
                return Mono.error(new InvalidRequestException(
                        "The transaction amount must be greater than zero"));
            }

            return getActiveCard(cardId)
                    .flatMap(card -> {
                        // 2. Validar TIPO: Si es débito, rechazar rápido con 422
                        if (!TYPE_CREDIT.equals(card.getType())) {
                            return Mono.error(new InvalidCardTypeException(
                                    "Consumptions apply only to CREDIT cards"));
                        }
                        // 3. Validar ESTADO: Si no está activa, rechazar con 422
                        if (!STATUS_ACTIVE.equals(card.getStatus())) {
                            return Mono.error(new InvalidCardStateException("Card is not ACTIVE"));
                        }
                        // 4. Validar SALDO: Si el monto es mayor, rechazar con 422
                        if (card.getAvailableBalance() == null || amount.compareTo(card.getAvailableBalance()) > 0) {
                            return Mono.error(new InsufficientCreditException(
                                    "Amount exceeds available credit line"));
                        }
                        // 5. Procesar atómicamente y registrar movimiento
                        return atomic.decrementAvailableBalance(cardId, amount)
                                .flatMap(updated -> updated
                                        ? recordMovement(cardId, "CONSUMPTION", amount, request)
                                        : Mono.error(new InsufficientCreditException(
                                        "Amount exceeds available credit line")));
                    })
                    .map(movementId -> {
                        OperationAcceptedResponse response = new OperationAcceptedResponse();
                        response.setOperationId(movementId);
                        response.setStatus("ACCEPTED");
                        return response;
                    });
        });
    }

    private Mono<UUID> recordMovement(UUID cardId, String type, BigDecimal amount, ConsumptionRequest request) {
        MovementDocument movement = MovementDocument.builder()
                .id(UUID.randomUUID())
                .cardId(cardId)
                .type(type)
                .amount(amount)
                .currency(request.getCurrency())
                .merchant(request.getMerchant())
                .timestamp(Instant.now())
                .build();
        return movementRepository.save(movement)
                .map(saved -> {
                    eventProducer.movementRegistered(saved).subscribe();
                    return saved.getId();
                });
    }

    @Override
    public Mono<BalanceResponse> findCardBalance(UUID cardId) {
        return getActiveCard(cardId)
                .map(card -> {
                    BalanceResponse response = new BalanceResponse();
                    response.setCardId(card.getId());
                    if (TYPE_CREDIT.equals(card.getType())) {
                        response.setAvailableBalance(card.getAvailableBalance().toPlainString());
                    } else {
                        response.setAvailableBalance(BigDecimal.ZERO.toPlainString());
                    }
                    response.setCurrency(card.getCurrency());
                    return response;
                });
    }

    @Override
    public Mono<MovementListResponse> findLastMovements(UUID cardId) {
        return getActiveCard(cardId)
                .thenMany(movementRepository.findTop10ByCardIdOrderByTimestampDesc(cardId))
                .collectList()
                .map(mapper::toMovementList);
    }

    @Override
    public Mono<EligibilityResponse> checkCardEligibility(UUID customerId) {
        return cardRepository.existsByCustomerIdAndTypeAndStatus(customerId, TYPE_CREDIT, STATUS_ACTIVE)
                .map(hasCard -> {
                    EligibilityResponse response = new EligibilityResponse();
                    response.setCustomerId(customerId);
                    response.setHasCreditCard(hasCard);
                    return response;
                });
    }

    private Mono<CardDocument> getActiveCard(UUID cardId) {
        return cardRepository.findByIdAndStatusNot(cardId, STATUS_CANCELLED)
                .switchIfEmpty(Mono.error(new CardNotFoundException("Card not found: " + cardId)));
    }
}
